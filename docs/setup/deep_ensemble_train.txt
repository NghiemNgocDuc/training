# ---------------------------------------------------------------------------
# 3. TRAIN THE 5 ENSEMBLE MEMBERS
#    Same train/val/test split for all members; ONLY the seed changes:
#    seeds = 42 123 7 2024 999
#    Everything else is byte-for-byte the verified fold-0 setup:

python aqm-spice2/freesolv/deep_ensemble.py --mode train --seeds 42 123 7 2024 999 --device cuda

#   split.md5                md5 of the frozen split (sanity anchor)
