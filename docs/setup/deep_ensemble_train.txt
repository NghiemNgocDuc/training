cd training
git pull

pip install torch torch_geometric h5py scikit-learn scipy matplotlib numpy rdkit

# ---------------------------------------------------------------------------
# 3. TRAIN + ANALYZE THE 5-MEMBER DEEP ENSEMBLE (runs in the background -
#    detach-friendly)
#
#    TRAIN: same train/val/test split for all members; ONLY the seed changes:
#    seeds = 42 123 7 2024 999. Everything else is byte-for-byte the verified
#    fold-0 setup: DimeNetPlus arch, lr=1e-4, wd=1e-5, batch=8, MSE-in-eV,
#    grad-clip 10.0, ReduceLROnPlateau(f=0.5, pat=15, min_lr=1e-6), early stop
#    pat=30 on val MAE, 5-conformer RDKit TTA. Run from anywhere - the script
#    chdirs into aqm-spice2/ itself.
#
#    The nohup '&' + output redirect keeps training alive after you close the
#    SSH/Jupyter terminal, so you can shut down your own computer.
#    (It does NOT survive a full instance restart / container reboot.)
#
#    ANALYZE: run after training finishes - sanity checks + UQ analysis.
# ---------------------------------------------------------------------------
nohup python aqm-spice2/freesolv/deep_ensemble.py --mode train --seeds 42 123 7 2024 999 --device cuda \
  > deep_ensemble_train.log 2>&1 &

# watch progress any time (Ctrl+C just stops the view, not the training):
tail -f deep_ensemble_train.log
